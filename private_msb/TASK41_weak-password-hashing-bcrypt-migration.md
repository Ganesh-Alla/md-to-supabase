# Weak Password Hashing Algorithm - Implement Bcrypt for Secure Storage

## Description

User passwords are stored using weak hashing (likely MD5 or SHA-1), making them vulnerable to brute-force and rainbow table attacks. Passwords should use bcrypt, which is specifically designed for password storage with built-in salt generation and configurable cost factor.

## Steps to Reproduce / Create the Issue

### To Identify Current Weak Implementation:

1. **Examine UserPassword entity**

**File:** `callRecordingHAR/src/net/voicelog/callRecording/entities/UserPassword.java`

Check how password is hashed:

```java
// WEAK PATTERNS to look for:
public void setPassword(String plainPassword) {
    this.password = MD5Util.hash(plainPassword);  // WEAK: MD5
    this.password = SHA1Util.hash(plainPassword);  // WEAK: SHA-1
    this.password = hashWithFixedSalt(plainPassword);  // WEAK: Fixed salt
}
```

2. **Check database schema**

```sql
-- View user_password table
SELECT * FROM user_password LIMIT 1;
-- Weak hashes appear as 32-char MD5 or 40-char SHA-1 hex strings
```

3. **Compare with bcrypt format**

```
Weak: 5f4dcc3b5aa765d61d8327deb882cf99  (MD5 of "12345")
Strong: $2b$12$E2E4UxYzh0AW8jFJHbVKOefAjMgXWXmKJkfH2qKkGm1zCFHCXBBwO  (bcrypt)
```

## Expected Behavior

Passwords should use bcrypt with:
- Unique random salt per password (automatically generated by bcrypt)
- Cost factor of 12 (takes ~100ms per hash)
- Constant-time verification (prevents timing attacks)
- Automatic migration from weak hashes on login

## Actual Behavior

If using MD5/SHA-1:
- All users with password "12345" have identical hashes
- Rainbow tables exist for billions of common passwords
- GPU attacks can crack 10+ billion passwords/second
- Database compromise exposes all passwords instantly

### Step 2: Create PasswordHasher Utility

**File to create:** `callRecording/src/net/voicelog/callRecording/security/PasswordHasher.java`

```java
import org.mindrot.jbcrypt.BCrypt;

public class PasswordHasher {

    private static final int BCRYPT_COST = 12;

    /**
     * Hash password using bcrypt
     */
    public static String hashPassword(String plainPassword) {
        if (plainPassword == null || plainPassword.isEmpty()) {
            throw new IllegalArgumentException("Password cannot be empty");
        }
        return BCrypt.hashpw(plainPassword, BCrypt.gensalt(BCRYPT_COST));
    }

    /**
     * Verify password with constant-time comparison
     */
    public static boolean verifyPassword(String plainPassword, String hashedPassword) {
        if (plainPassword == null || hashedPassword == null) {
            return false;
        }
        try {
            return BCrypt.checkpw(plainPassword, hashedPassword);
        } catch (IllegalArgumentException e) {
            return false;
        }
    }

    /**
     * Check if password uses old weak format
     */
    public static boolean isWeakHash(String hash) {
        if (hash == null) {
            return true;
        }
        // Bcrypt hashes start with $2a$, $2b$, or $2y$
        return !hash.startsWith("$2a$") &&
               !hash.startsWith("$2b$") &&
               !hash.startsWith("$2y$");
    }
}
```

### Step 4: Update AdminSessionBean

**File to modify:** `callRecording/src/net/voicelog/callRecording/ejb/AdminSessionBean.java`

```java
public User authenticateUser(String username, String plainPassword) {
    User user = getUserByUsername(username);
    if (user == null) {
        return null;
    }

    UserPassword userPassword = user.getUserPassword();

    // Verify password
    if (!PasswordHasher.verifyPassword(plainPassword, userPassword.getPassword())) {
        return null;
    }

    // AUTO-UPGRADE: If using old weak hash, upgrade on login
    if (PasswordHasher.isWeakHash(userPassword.getPassword())) {
        String newHash = PasswordHasher.hashPassword(plainPassword);
        userPassword.setPassword(newHash);
        getSession().update(userPassword);
        getSession().flush();

        AuditInformationHelper.logAudit(
            "PASSWORD_UPGRADED",
            user.getId(),
            "Password hash upgraded from weak to bcrypt on login"
        );
    }

    return user;
}

public void changePassword(User user, String oldPassword, String newPassword) {
    UserPassword userPassword = user.getUserPassword();

    if (!PasswordHasher.verifyPassword(oldPassword, userPassword.getPassword())) {
        throw new InvalidPasswordException("Old password incorrect");
    }

    userPassword.setPassword(newPassword);  // Uses bcrypt
    getSession().update(userPassword);

    AuditInformationHelper.logAudit(
        "PASSWORD_CHANGED",
        user.getId(),
        "User changed password"
    );
}
```

```java
public class PasswordHasherTest extends TestCase {

    /**
     * Test bcrypt hashing creates different hashes for same password
     */
    public void testBcryptUniqueHashes() {
        String password = "MySecureP@ssword";

        String hash1 = PasswordHasher.hashPassword(password);
        String hash2 = PasswordHasher.hashPassword(password);

        // Bcrypt generates different hash each time (different salt)
        assertNotEquals("Hashes should be different due to random salt",
            hash1, hash2);

        // But both should verify against same password
        assertTrue(PasswordHasher.verifyPassword(password, hash1));
        assertTrue(PasswordHasher.verifyPassword(password, hash2));
    }

    /**
     * Test bcrypt format validation
     */
    public void testBcryptFormatValidation() {
        String password = "MyPassword";
        String hash = PasswordHasher.hashPassword(password);

        // Should start with bcrypt prefix
        assertTrue("Hash should start with $2",
            hash.startsWith("$2a$") || hash.startsWith("$2b$") || hash.startsWith("$2y$"));

        // Should have cost factor
        assertTrue("Hash should contain cost factor",
            hash.contains("$12$"));
    }

    /**
     * Test password verification
     */
    public void testPasswordVerification() {
        String password = "CorrectPassword123";
        String hash = PasswordHasher.hashPassword(password);

        assertTrue(PasswordHasher.verifyPassword(password, hash));
        assertFalse(PasswordHasher.verifyPassword("WrongPassword", hash));
        assertFalse(PasswordHasher.verifyPassword("", hash));
        assertFalse(PasswordHasher.verifyPassword(null, hash));
    }

    /**
     * Test weak hash detection
     */
    public void testWeakHashDetection() {
        String md5Hash = "5f4dcc3b5aa765d61d8327deb882cf99";  // MD5 of "12345"
        String sha1Hash = "356a192b7913b04c54574d18c28d46e6395428ab";  // SHA-1 of "12345"
        String bcryptHash = PasswordHasher.hashPassword("12345");

        assertTrue("Should detect MD5 as weak", PasswordHasher.isWeakHash(md5Hash));
        assertTrue("Should detect SHA-1 as weak", PasswordHasher.isWeakHash(sha1Hash));
        assertFalse("Should not detect bcrypt as weak", PasswordHasher.isWeakHash(bcryptHash));
    }

    /**
     * Test constant-time comparison (timing attack protection)
     */
    public void testConstantTimeComparison() {
        String password = "MyPassword";
        String hash = PasswordHasher.hashPassword(password);

        long[] times = new long[100];

        // Measure time for wrong password
        for (int i = 0; i < 100; i++) {
            long start = System.nanoTime();
            PasswordHasher.verifyPassword("wrong" + i, hash);
            times[i] = System.nanoTime() - start;
        }

        // Times should be relatively consistent (constant-time)
        long minTime = Long.MAX_VALUE, maxTime = 0;
        for (long t : times) {
            minTime = Math.min(minTime, t);
            maxTime = Math.max(maxTime, t);
        }

        long variance = maxTime - minTime;
        // Should have <50% variance (timing attacks much harder)
        assertTrue("Timing variance too high, possible attack vector",
            variance < (minTime * 0.5));
    }

    /**
     * Test null and empty handling
     */
    public void testNullAndEmptyHandling() {
        assertThrows(IllegalArgumentException.class, () ->
            PasswordHasher.hashPassword(null)
        );
        assertThrows(IllegalArgumentException.class, () ->
            PasswordHasher.hashPassword("")
        );

        assertFalse(PasswordHasher.verifyPassword(null, "hash"));
        assertFalse(PasswordHasher.verifyPassword("password", null));
    }
}
```

### Integration Test: PasswordMigrationTest.java

**Location:** `callRecording/test/net/voicelog/test/integration/PasswordMigrationTest.java`

```java
public class PasswordMigrationTest extends TestCase {

    private Client client;
    private AdminSessionBean adminSessionBean;

    @Override
    protected void setUp() throws Exception {
        super.setUp();
        client = Utility.getTestSessionBean()
            .findClientByName(BootstrapConstants.clientNameOne);
        adminSessionBean = Utility.getAdminSessionBean();
    }

    /**
     * Test that weak password hash is upgraded on login
     */
    public void testWeakPasswordUpgradeOnLogin() throws Exception {
        // Create user with simulated weak password (MD5)
        User user = new User();
        user.setUserName("testuser_weak");
        user.setFirstName("Test");
        user.setLastName("User");
        user.setClient(client);

        Long userId = adminSessionBean.createUser(user, "TestPassword123");

        // Simulate weak hash in database
        UserPassword userPassword = user.getUserPassword();
        String weakHash = "5f4dcc3b5aa765d61d8327deb882cf99";  // MD5 of "12345"
        userPassword.setPassword(weakHash);  // Force weak hash

        // Attempt login
        User authenticatedUser = adminSessionBean.authenticateUser(
            "testuser_weak",
            "TestPassword123"
        );

        // After successful login, password should be upgraded
        UserPassword updatedPassword = authenticatedUser.getUserPassword();
        assertFalse("Password should be upgraded from weak hash",
            PasswordHasher.isWeakHash(updatedPassword.getPassword()));
        assertTrue("Updated password should be bcrypt",
            updatedPassword.getPassword().startsWith("$2"));
    }

    /**
     * Test password change uses bcrypt
     */
    public void testPasswordChangeUsesBcrypt() throws Exception {
        User user = Utility.getTestSessionBean()
            .findUserByUserName(client, BootstrapConstants.adminUserName);
        String oldPassword = BootstrapConstants.adminUserPassword;
        String newPassword = "NewSecureP@ssword123";

        adminSessionBean.changePassword(user, oldPassword, newPassword);

        // Verify new password is bcrypt hash
        UserPassword userPassword = user.getUserPassword();
        assertFalse("Should not be weak hash",
            PasswordHasher.isWeakHash(userPassword.getPassword()));
        assertTrue(PasswordHasher.verifyPassword(newPassword, userPassword.getPassword()));
    }

    /**
     * Test multiple users with same password have different hashes
     */
    public void testUniqueHashPerPassword() throws Exception {
        String samePassword = "SamePassword123";

        // Create two users with same password
        User user1 = new User();
        user1.setUserName("user_1");
        user1.setClient(client);
        Long id1 = adminSessionBean.createUser(user1, samePassword);

        User user2 = new User();
        user2.setUserName("user_2");
        user2.setClient(client);
        Long id2 = adminSessionBean.createUser(user2, samePassword);

        // Get both users from database
        User savedUser1 = Utility.getTestSessionBean().findUserById(id1);
        User savedUser2 = Utility.getTestSessionBean().findUserById(id2);

        String hash1 = savedUser1.getUserPassword().getPassword();
        String hash2 = savedUser2.getUserPassword().getPassword();

        // Hashes should be DIFFERENT even though password is same
        assertNotEquals("Same password should produce different bcrypt hashes",
            hash1, hash2);

        // Both should still verify
        assertTrue(PasswordHasher.verifyPassword(samePassword, hash1));
        assertTrue(PasswordHasher.verifyPassword(samePassword, hash2));
    }
}
```
